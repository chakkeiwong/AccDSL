def dump(path,value):path.write_text(json.dumps(value,indent=2)+chr(10))
def sha(path):return sha256(path.read_bytes()).hexdigest()

def packet(records, selections, context=None):
    context={'bank':'Synthetic-US-Bank','book':'Legal','perimeter':'Solo'} if context is None else context
    definition={'profile':'MethodRequirements/1','method_id':'timestamp.selection-only','version':'1',
                'inputs':{},'requirements':[],'capabilities':[],
                'residual_prerequisites':['Binding declarations only; timestamp conversion is separate.']}
    schemas,refs,bindings,blobs={},{},[],{}
    for name,record in records.items():
        raw=core.canonical(record);key=sha256(raw).hexdigest();blobs[key]=raw
        schema={'profile':binder.SCHEMA,'requirements_profile':'MethodRequirements/1',
                'schema_id':'selected-time.'+name,'fields':{}}
        for input_id,origin,pointer,encoding,role in selections:
            if origin!=name:continue
            typ={'kind':'text','context':context.copy()}
            definition['inputs'][input_id]={'type':typ,'role':role,'source_ref':'sha256:'+key+'#'+pointer}
            schema['fields'][pointer]={'type':typ,'role':role,'encoding':encoding,'missing':'unresolved'}
            bindings.append({'input':input_id,'record':name,'pointer':pointer})
            definition['requirements'].append({'id':'selected.'+input_id,'predicate':{'op':'eq','left':{'op':'ref','id':input_id},'right':{'op':'ref','id':input_id}},
                 'requires':[input_id],'admission_refs':[],'on_false':'invalid','on_unknown':'unresolved'})
        sk=core.digest(schema);schemas[sk]=schema;refs[name]={'raw_sha256':key,'schema_digest':sk}
    dk=core.digest(definition)
    return {'plan':{'profile':binder.PROFILE,'manifest_digest':dk,'records':refs,'bindings':bindings},
            'manifests':{dk:definition},'schemas':schemas,'blobs':blobs,'available_capabilities':[]}

def pair_packet(a,b):
    return packet({'supplied':{'a':a,'b':b}},[(x,'supplied','/'+x,'tagged_binding','caller_finding') for x in ['a','b']])

def compare(projected):
    assert projected['status']=='projected'
    record=projected['record'];raw=core.canonical(record);key=sha256(raw).hexdigest()
    definition={'profile':'MethodRequirements/2','method_id':'timestamp.available-by','version':'1',
                'inputs':{},'requirements':[{'id':'available-by','predicate':{'op':'le','left':{'op':'ref','id':'a'},'right':{'op':'ref','id':'b'}},
                'requires':['a','b'],'admission_refs':[],'on_false':'invalid','on_unknown':'unresolved'}],
                'capabilities':[],'residual_prerequisites':['Evidence membership and timestamp truth remain unassessed.']}
    schema={'profile':binder.SCHEMA,'requirements_profile':'MethodRequirements/2','schema_id':'exact-projected-times','fields':{}}
    bindings=[]
    for name in ['a','b']:
        row=record['values'][name];pointer='/values/'+name+'/value'
        definition['inputs'][name]={'type':row['type'],'role':row['role'],'source_ref':'sha256:'+key+'#'+pointer}
        schema['fields'][pointer]={'type':row['type'],'role':row['role'],'encoding':'tagged_binding','missing':'unresolved'}
        bindings.append({'input':name,'record':'projection','pointer':pointer})
    dk,sk=core.digest(definition),core.digest(schema)
    args={'plan':{'profile':binder.PROFILE,'manifest_digest':dk,'records':{'projection':{'raw_sha256':key,'schema_digest':sk}},'bindings':bindings},
          'manifests':{dk:definition},'schemas':{sk:schema},'blobs':{key:raw},'available_capabilities':[]}
    result=binder.assess_bound(**args);assert result['status']=='bound'
    exchange={'profile':text.EXCHANGE,'definition':definition,'selected_digest':dk,'values':result['values'],'available_capabilities':[]}
    rendered=text.render(exchange);assert text.lower(rendered)==exchange
    assert text.assess_exchange(text.lower(rendered))==result['assessment']
    for obj in [projected,result,result['assessment']]:
        assert all(obj[k] is False for k in BOUNDARY)
    return result,rendered,args

def known(x):return {'status':'known','value':x}

def manual_days(y,m,d):
    previous=y-1
    days=365*previous+previous//4-previous//100+previous//400
    leap=y%4==0 and (y%100!=0 or y%400==0)
    months=[31,29 if leap else 28,31,30,31,30,31,31,30,31,30,31]
    return days+sum(months[:m-1])+d-1
